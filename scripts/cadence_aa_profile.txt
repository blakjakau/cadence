abi <abi/4.0>,
include <tunables/global>

# =========================================================================
# 1. HOST SPAWNER (cadence)
# Grants 100% of normal user permissions + userns creation.
# Normal user PTY children inherit this permissive state or run unconfined.
# =========================================================================
profile cadence /**/cadence{-dev,-latest,-linux-x64,-linux-arm64} flags=(attach_disconnected, mediate_deleted) {
  userns,
  capability,
  network,
  mount,
  remount,
  umount,
  pivot_root,
  ptrace,
  signal,
  mqueue,
  unix,
  dbus,
  io_uring,

  # Full read, write, lock, link, mmap, and inherit-exec on all files.
  # Because this profile allows everything ABOVE, any command you run in
  # the host user PTY (bash, ls, git, ssh) that inherits this profile
  # has zero restrictions!
  allow file rwlkmix /{**,},

  # Allow programmatic transition into the sandbox keeper (and shells)
  change_profile -> cadence_sbx_keeper,
  change_profile -> cadence_sbx_shell,
  change_profile -> cadence_sbx_net,
  change_profile -> cadence,
}

# =========================================================================
# 2. SANDBOX KEEPER (cadence_sbx_keeper)
# Entered via writing "exec cadence_sbx_keeper" to /proc/self/attr/apparmor/exec
# =========================================================================
profile cadence_sbx_keeper flags=(attach_disconnected, mediate_deleted) {
  deny capability,
  network,
  unix,
  signal,
  ptrace (read, trace) peer=cadence_sbx_shell,
  ptrace (read, trace) peer=cadence_sbx_net,

  # Full file rwlkm + inherit-exec inside its mount namespace
  allow file rwlkmix /{**,},

  # Allow keeper to transition spawned children into the sandbox shells
  change_profile -> cadence_sbx_shell,
  change_profile -> cadence_sbx_net,
}

# =========================================================================
# 3. OFFLINE AGENT SHELL (cadence_sbx_shell)
# =========================================================================
profile cadence_sbx_shell flags=(attach_disconnected, mediate_deleted) {
  # Hard blocks
  deny capability,
  deny network,
  deny dbus,
  deny io_uring,
  deny mount,
  deny umount,
  deny pivot_root,
  deny ptrace,

  # Allow signals and anonymous/local pipes & unix sockets inside the sandbox
  signal (send, receive) peer=cadence_sbx_shell,
  signal (send, receive) peer=cadence_sbx_keeper,
  signal (send, receive) peer=cadence,
  unix,

  # Tool allowlist + project writes. AppArmor is deny-by-default, so a
  # blanket exec grant is what let curl/python run even in an air-gapped
  # shell. The old "rwlkmix" was rwlkm + ix, where ix = "inherit execute"
  # (the exec'd program inherits THIS profile). A bare x is only legal in
  # deny rules, so the fix is to drop the exec transition from the blanket
  # baseline (rwlkm = read/write/mmap/lock/link on whatever the mount
  # namespace exposes = the rw project roots) and grant ix ONLY on the
  # standard tool dirs. bash/curl/etc. in /usr still run (inheriting this
  # profile), but a binary that lands in a project root or any non-tool dir
  # cannot be executed. The mount namespace hides non-tool, non-root paths
  # anyway, so this is defense-in-depth on top of the mount isolation.
  # Sensitive host paths are explicitly blocked even if shared in (below).
  allow file rwlkm /{**,},
  allow file ix /usr/{**,},
  allow file ix /lib/{**,},
  allow file ix /lib64/{**,},
  allow file ix /bin/{**,},
  allow file ix /sbin/{**,},
  allow file ix /opt/{**,},
  allow file ix /proc/{**,},
  allow file ix /etc/{**,},
  # Project roots live under the user's home (@{HOME} = /home/*). The mount
  # namespace exposes only the configured rw roots, so this grant is bounded
  # to the actual project dirs. It lets the user run their own build scripts,
  # test binaries, and tools from the project tree -- the primary use case.
  allow file ix @{HOME}/{**,},

  deny @{HOME}/.ssh/** rwklx,
  deny @{HOME}/.gnupg/** rwklx,
  deny @{HOME}/.aws/** rwklx,
  deny @{HOME}/.config/gcloud/** rwklx,
  deny @{HOME}/.bashrc wkl,
  deny @{HOME}/.profile wkl,
  deny @{HOME}/.zshrc wkl,
  deny @{HOME}/.config/systemd/** rwklx,
  deny /run/user/*/** rwklx,
  deny /var/run/docker.sock rwklx,
}

# =========================================================================
# 4. NETWORKED AGENT SHELL (cadence_sbx_net)
# =========================================================================
profile cadence_sbx_net flags=(attach_disconnected, mediate_deleted) {
  deny capability,
  deny dbus,
  deny io_uring,
  deny mount,
  deny umount,
  deny pivot_root,
  deny ptrace,

  # Network allowed in this profile
  network inet,
  network inet6,
  network netlink raw,

  signal (send, receive) peer=cadence_sbx_net,
  signal (send, receive) peer=cadence_sbx_keeper,
  signal (send, receive) peer=cadence,
  unix,

  # Same deny-then-allow tool allowlist as the offline shell (cadence_sbx_shell).
  # A networked shell is strictly more dangerous than an air-gapped one: an
  # exec'd binary here can reach out over the net, so the blanket exec grant
  # would let a tool dropped into a project root phone home. rwlkm (no ix) on
  # the mount-namespace exposure, ix on the standard tool dirs AND the project
  # tree (the mount ns bounds it to the configured rw roots).
  allow file rwlkm /{**,},
  allow file ix /usr/{**,},
  allow file ix /lib/{**,},
  allow file ix /lib64/{**,},
  allow file ix /bin/{**,},
  allow file ix /sbin/{**,},
  allow file ix /opt/{**,},
  allow file ix /proc/{**,},
  allow file ix /etc/{**,},
  # Project roots live under the user's home (@{HOME} = /home/*). The mount
  # namespace exposes only the configured rw roots, so this grant is bounded
  # to the actual project dirs. It lets the user run their own build scripts,
  # test binaries, and tools from the project tree -- the primary use case.
  allow file ix @{HOME}/{**,},

  deny @{HOME}/.ssh/** rwklx,
  deny @{HOME}/.gnupg/** rwklx,
  deny @{HOME}/.aws/** rwklx,
  deny @{HOME}/.config/gcloud/** rwklx,
  deny @{HOME}/.bashrc wkl,
  deny @{HOME}/.profile wkl,
  deny @{HOME}/.zshrc wkl,
  deny @{HOME}/.config/systemd/** rwklx,
  deny /run/user/*/** rwklx,
  deny /var/run/docker.sock rwklx,
}
